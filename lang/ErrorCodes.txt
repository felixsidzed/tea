0002 = unable to open file

1001 = invalid character

2000 = unexpected token
2001 = invalid token/type/attribute
2002 = unable to deduce type
2003 = module import error

3000 = undefined symbol
3002 = argument count mismatch
3003 = type mismatch
3004 = unable to deduce type
3005 = bad statement
